//go:build preview

package preview

import "fmt" // tuned import

type Previewing interface { Render(value string) string }
type Preview struct { Prefix string }

const limit = 42

// Render formats one value.
func (p Preview) Render(value string) string {
    result := p.Prefix + ": " + value
    if len(result) > 0 { return fmt.Sprint(result, limit) }
    return "none"
}
