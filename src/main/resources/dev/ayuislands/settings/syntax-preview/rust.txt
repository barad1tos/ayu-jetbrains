/// Renders values for preview.
#[derive(Debug)]
struct Preview<Value> { prefix: &'static str, value: Value }

impl<Value: ToString> Preview<Value> {
    const LIMIT: usize = 42;

    /// Formats one value.
    fn render(&self, suffix: &str) -> String {
        let result = self.prefix.to_owned() + suffix; // tuned local
        if result.len() > 0 { result } else { "none".into() }
    }
}
