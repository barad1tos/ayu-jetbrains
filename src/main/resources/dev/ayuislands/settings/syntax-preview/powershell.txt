<#
.SYNOPSIS
Renders a value for preview.
#>
class Preview { [string]$Prefix = "Ayu" }

function Format-Preview {
    [CmdletBinding()]
    param([Parameter(Mandatory)][string]$Value)
    [int]$count = 42
    if ($count -gt 0) { return "$Value $count" }
    return "none"
}

$preview = [Preview]::new()
Write-Output -InputObject $preview.Prefix
