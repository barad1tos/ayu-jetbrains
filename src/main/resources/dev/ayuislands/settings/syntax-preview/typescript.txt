/** Renders values for preview. */
interface Previewing<Value> {
    render(value: Value): string;
}

@sealed
class Preview<Value> implements Previewing<Value> {
    private prefix: string = "Ayu";

    /** Formats one value. */
    render(value: Value): string {
        const result: string = `${this.prefix}: ${value}`; // tuned local
        return result;
    }
}
